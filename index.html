HospitalSystem.java
public class HospitalSystem {
  // Patient Registration & Scheduling
  private List<Patient> patients;
  private Map<Doctor, List<Appointment>> schedule;

  public void registerPatient(Patient p) {
    patients.add(p);
    System.out.println("Patient registered: " + p.getId());
  }

  public Appointment scheduleSlot(Patient p, Doctor d, Date time) {
    Appointment appt = new Appointment(p, d, time);
    schedule.get(d).add(appt);
    return appt;
  }
}
